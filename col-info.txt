function render() {
    if (!document.body.classList.contains('desktop-mode') && document.getElementById('chartOverlay').style.display === 'flex') return;

    const currentPairs = filteredPairs.length ? filteredPairs : PAIRS;
    let filtered;
    const validTargets = new Set(getUniqueAutoSetups().keys());
    if (curF === 'all') filtered = currentPairs;
    else if (curF === 'bull' || curF === 'bear')
        filtered = currentPairs.filter(p => {
            const d = MARKET_DATA[p.n] || {};
            return ['4h','1day','1week'].every(tf => d[tf] === curF);
        });
    else if (curF === 'target') filtered = currentPairs.filter(p => validTargets.has(p.n));
    else filtered = currentPairs;

    if (currentSLevel) {
        const SLEVEL_HTF_MAP = { H: 'W', M: '240', L: '30' };
        const sLevelHtfKey = SLEVEL_HTF_MAP[currentTFPreset] || 'W';
        const sLevelWantDir = currentDirection === 'long' ? 'bull' : 'bear';
        filtered = filtered.filter(p => {
            const s = (SLEVEL_STATE[sLevelHtfKey] || {})[p.n];
            return s && s.direction === sLevelWantDir && s.level === currentSLevel;
        });
        if (!sortColumn) {
            const moveCol = { type: 'change_200' };
            filtered = [...filtered].sort((a, b) => {
                const va = sortValueFor(a.n, moveCol);
                const vb = sortValueFor(b.n, moveCol);
                const aAbs = va == null ? -Infinity : Math.abs(va);
                const bAbs = vb == null ? -Infinity : Math.abs(vb);
                return bAbs - aAbs;
            });
        }
    }

    filtered = applyColumnSortAndFilter(filtered);
    currentDisplayPairs = filtered.map(p => p.n);
    if (window.activeChartSource !== 'watchlist') { chartPairs = filtered.map(p => p.n); }

    renderHeader();

    document.getElementById('pairsContainer').innerHTML = filtered.map(p => {
        let cells = `<div class="col-pair-fixed"><span style="font-weight:700;font-size:14px;">${p.n}</span></div>`;
        cells += columnDefs.map(col => renderColumnCell(p.n, col)).join('');
        cells += '<div class="col-add-btn" style="visibility:hidden;"></div>';

        let row = `<div class="item-row ${p.n === activeChartPair ? 'highlighted-row' : ''}"><div class="item-main" onclick="openChartForPair('${p.n}')">${cells}</div>`;
        if (curF === 'target') {
            const info = getUniqueAutoSetups().get(p.n);
            if (info) {
                const badgeBg = info.dir === 'bull' ? '#22c55e' : '#ef4444';
                row += `<div class="phase-row"><span style="color:var(--muted);">Phase:</span><span style="background:${badgeBg};color:#fff;padding:2px 10px;border-radius:4px;font-size:10px;font-weight:700;">${info.dir.toUpperCase()} ${info.phase}</span></div>`;
            }
        }
        row += '</div>';
        return row;
    }).join('');
    updateCounts();
}
